The compiler refuses it, because a broader catch block placed before a narrower one makes the narrower one unreachable.

public class CatchOrderError {
    public static void main(String[] args) {
        try {
            Integer.parseInt("abc");
        } catch (Exception e) {                    // broad catch first
            System.out.println("Exception");
        } catch (NumberFormatException e) {        // ERROR: exception NumberFormatException has already been caught
            System.out.println("NumberFormatException");
        }
    }
}
